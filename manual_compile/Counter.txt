class Counter

    field int value, step;

    constructor Counter new (int startValue, int s)
        let value = startValue
        let step = s
        return this
    
    method void inc()
        let value = value + step

    method int get()
        return value

    method int doubled()
        return get() * 2

    function void demo()
        var Counter c
        let c = Counter.new(0, 5)
        do c.inc()
        do Output.printInt(c.get())
        return
