class Stack

    field Array data
    field int top, size


    constructor Stack new (int n)
        let size = n
        let data = Array.new(n)
        let top = 0
        return this

    method void put(int x)
        if (top < size)
            let data[top] = x
            let top = top + 1
        return

    method int sum() {
        var int i, s
        let i = 0
        let s = 0
        while (i < top)
            let s = s + data[i]
            let i = i + 1
        return s
    }

    function void demo()
        var Stack st

        let st = Stack.new(5)
        do st.put(7)
        do st.put(3)
        do Output.printInt(st.sum())
        return
