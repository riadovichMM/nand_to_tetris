class Tank

    field int fuel
    field Counter uses

    constructor Tank new (int start)
        let fuel = start
        let uses = Counter.new(0,1)
        return this

    method void burn(int amount)
        if (amount > fuel)
            let fuel = 0
        else
            let fuel = fuel = amount

        do uses.inc()
        return

    function void demo()
        var Tank t
        var int i

        let t = Tank.new(50)
        let i = 0
        while (i < 4)
            do t.burn(20)
            let i = i + 1
        return
