class Wallet

    field int balance
    field Counter ops


    counstructor Wallet new(int start)
        let balance = start
        let ops = Counter.new(0, 1)
        return this


    method boolean withdraw(int amount)
        if (amount > balance)
            return false
        let balance = balance - amount
        do ops.inc()
        return true

    function void demo()
        var Wallet w
        var int i
        let w = Wallet.new(100)
        let i = 0
        while (i < 3)
            do w.withdraw(10)
            let i = i + 1
        return
